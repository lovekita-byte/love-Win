    <script>
        const correctPassword = "1234"; // CHANGE YOUR PASSWORD HERE
        let currentQuestionIndex = 0;
        let score = 0;
        
        const questions = [
            { q: "When is our anniversary?", options: ["Jan 1", "Feb 14", "March 15", "May 20"], answer: 2 },
            { q: "What is my favorite color?", options: ["Red", "Blue", "Pink", "Purple"], answer: 2 },
            { q: "Where did we first meet?", options: ["Mall", "School", "Park", "Church"], answer: 1 },
            { q: "What is my favorite food?", options: ["Pizza", "Burgers", "Pasta", "Sushi"], answer: 0 },
            { q: "What is my favorite movie?", options: ["Romance", "Action", "Comedy", "Horror"], answer: 0 },
            { q: "What is my favorite song?", options: ["Love Song", "Pop", "Rock", "Classical"], answer: 0 },
            { q: "What is my favorite flower?", options: ["Rose", "Tulip", "Sunflower", "Lily"], answer: 0 },
            { q: "What is my favorite drink?", options: ["Coke", "Water", "Juice", "Coffee"], answer: 1 },
            { q: "What is my favorite sport?", options: ["Basketball", "Volleyball", "Soccer", "Tennis"], answer: 0 },
            { q: "What is my favorite season?", options: ["Summer", "Rainy", "Winter", "Spring"], answer: 0 },
        ];

        function createHearts() {
            const container = document.getElementById('hearts-container');
            for (let i = 0; i < 20; i++) {
                const heart = document.createElement('div');
                heart.className = 'bg-heart';
                heart.style.left = Math.random() * 100 + 'vw';
                heart.style.animationDuration = (Math.random() * 5 + 5) + 's';
                heart.style.animationDelay = Math.random() * 5 + 's';
                container.appendChild(heart);
            }
        }
        createHearts();

        function checkPassword() {
            const input = document.getElementById('password').value;
            if (input === correctPassword) {
                document.getElementById('lockContainer').classList.remove('active');
                document.getElementById('quizContainer').classList.add('active');
                loadQuestion();
            } else {
                const lockBody = document.querySelector('#lockContainer .lock-body');
                lockBody.classList.add('shake');
                setTimeout(() => lockBody.classList.remove('shake'), 500);
            }
        }

        function loadQuestion() {
            if (currentQuestionIndex >= questions.length) {
                showResult();
                return;
            }
            const q = questions[currentQuestionIndex];
            document.getElementById('currentQ').innerText = currentQuestionIndex + 1;
            document.getElementById('score').innerText = score;
            document.getElementById('questionText').innerText = q.q;
            const container = document.getElementById('optionsContainer');
            container.innerHTML = '';
            q.options.forEach((opt, index) => {
                const btn = document.createElement('div');
                btn.className = 'option';
                btn.innerText = opt;
                btn.onclick = () => selectAnswer(index);
                container.appendChild(btn);
            });
        }

        function selectAnswer(index) {
            const correctIndex = questions[currentQuestionIndex].answer;
            if (index === correctIndex) score++;
            currentQuestionIndex++;
            loadQuestion();
        }

        function showResult() {
            document.getElementById('quizContainer').classList.remove('active');
            document.getElementById('loveModal').style.display = 'flex';
            document.getElementById('finalScore').innerText = score;
        }

        function retryQuiz() {
            currentQuestionIndex = 0;
            score = 0;
            document.getElementById('failContainer').classList.remove('active');
            document.getElementById('lockContainer').classList.add('active');
        }

        function restartAll() {
            currentQuestionIndex = 0;
            score = 0;
            document.getElementById('failContainer').classList.remove('active');
            document.getElementById('quizContainer').classList.remove('active');
            document.getElementById('lockContainer').classList.add('active');
            document.getElementById('password').value = '';
        }

        function closeModal() {
            document.getElementById('loveModal').style.display = 'none';
        }
    </script>
